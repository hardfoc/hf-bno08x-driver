63inline std::size_t EncodeFrame(const uint8_t* shtp, std::size_t len, uint8_t* out,
64 std::size_t cap) noexcept {
65 std::size_t n = 0;
66 auto put = [&](uint8_t b) {
67 if (n < cap) {
68 out[n] = b;
69 }
70 ++n;
71 };
72 put(FLAG);
73 put(PROTOCOL_SHTP);
74 for (std::size_t i = 0; i < len; ++i) {
75 const uint8_t b = shtp[i];
76 if (b == FLAG || b == ESCAPE) {
77 put(ESCAPE);
78 put(static_cast<uint8_t>(b ^ ESCAPE_XOR));
79 } else {
80 put(b);
81 }
82 }
83 put(FLAG);
84 return n <= cap ? n : 0;
85}

110 Event Push(uint8_t c) noexcept {
111 switch (state_) {
112 case State::Outside:
113 if (c == FLAG) {
114 startFrame();
115 }
116 return Event::None;
117 case State::Escaped:
118 store(static_cast<uint8_t>(c ^ ESCAPE_XOR));
119 state_ = State::Inside;
120 return Event::None;

123 }
124 if (c == ESCAPE) {

126 return Event::None;
127 }
128 if (c != FLAG) {

131 }

133 const Event ev = finishFrame();

136 }

176 void store(uint8_t c) noexcept {
177 if (len_ < Capacity) {
178 frame_[len_++] = c;
179 } else {
180 overflow_ = true;
181 }

184 Event finishFrame() noexcept {
185 if (len_ == 0) {
186 return Event::None; // two flags in a row: an empty frame, or a closing flag reused
187 }
188 if (overflow_) {
189 return Event::Dropped;
190 }
191 if (frame_[0] == PROTOCOL_SHTP) {
192 shtpLen_ = len_ - 1;
193 return Event::ShtpFrame;
194 }
195 if (frame_[0] == PROTOCOL_CONTROL && len_ >= 3) {
196 bufferSpace_ = static_cast<uint16_t>(frame_[1] | (frame_[2] << 8));
197 return Event::BufferStatus;
198 }
199 return Event::Dropped;
200 }
