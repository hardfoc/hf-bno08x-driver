39template <typename CommType>
40bool BNO085<CommType>::Begin() noexcept {
41 if (state_ == BNO085DriverState::Sh2Active)
42 return true;
43 if (state_ != BNO085DriverState::Closed) {
44 last_error_ = SH2_ERR_OP_IN_PROGRESS;
45 return false;
46 }
47 if (io_.GetInterfaceType() == BNO085Interface::UARTRVC) {
48 last_error_ = SH2_ERR_BAD_PARAM;
49 return false;
50 }
51
52 // The SH-2 library has one global session; a second sh2_open() would reset the first.
53 if (!acquireSh2Session()) {
54 last_error_ = BNO08X_ERR_SH2_SESSION_IN_USE;
55 return false;
56 }
57
58 last_error_ = 0;
59 prepareHalWrapper();
60
61 int status = sh2_open(halWrapper_.asHal(), asyncCallback, this);
62 if (status != SH2_OK) {
63 last_error_ = status;
64 io_.Close();
65 releaseSh2Session();
66 return false;
67 }
68 status = sh2_reinitialize();
69 if (status != SH2_OK) {
70 last_error_ = status;
71 sh2_close();
72 io_.Close();
73 releaseSh2Session();
74 return false;
75 }
76 status = sh2_setSensorCallback(sensorCallback, this);
77 if (status != SH2_OK) {
78 last_error_ = status;
79 sh2_close();
80 io_.Close();
81 releaseSh2Session();
82 return false;
83 }
84
85 new_flag_.fill(false);
86 last_interval_.fill(0);
87 last_sensitivity_.fill(0);
88 state_ = BNO085DriverState::Sh2Active;
89 return true;
90}

99template <typename CommType>
100bool BNO085<CommType>::EnableSensor(BNO085Sensor sensor, uint32_t interval_ms,
101 float sensitivity) noexcept {
102 if (state_ != BNO085DriverState::Sh2Active) {
103 last_error_ = SH2_ERR_OP_IN_PROGRESS;
104 return false;
105 }
106 auto id = static_cast<uint8_t>(sensor);
107 if (id >= last_interval_.size())
108 return false;
109 uint32_t interval_us = interval_ms * 1000;
110 if (!configure(sensor, interval_us, sensitivity, 0))
111 return false;
112 last_interval_[id] = interval_us;
113 last_sensitivity_[id] = sensitivity;
114 return true;
115}

537template <typename CommType>
538void BNO085<CommType>::handleSensorEvent(const sh2_SensorEvent_t* event) noexcept {
539 sh2_SensorValue_t value;
540 int decode_status = sh2_decodeSensorEvent(&value, event);
541 if (decode_status != SH2_OK) {
542 last_error_ = decode_status;
543 return;
544 }
545 uint8_t id = value.sensorId;
546 if (id >= latest_.size())
547 return;
548 latest_[id] = value;
549 new_flag_[id] = true;
550 if (callback_) {
551 callback_(toSensorEvent(static_cast<BNO085Sensor>(id), value));
552 }
553}

561template <typename CommType>
562void BNO085<CommType>::handleAsyncEvent(const sh2_AsyncEvent_t* event) noexcept {
563 if (event->eventId == SH2_RESET) {
564 for (uint8_t id = 0; id < last_interval_.size(); ++id) {
565 if (last_interval_[id])
566 configure(static_cast<BNO085Sensor>(id), last_interval_[id], last_sensitivity_[id], 0);
567 }
568 }

580template <typename CommType>
581bool BNO085<CommType>::configure(BNO085Sensor sensor, uint32_t interval_us, float sensitivity,
582 uint32_t batch_us) noexcept {
583 sh2_SensorConfig_t cfg{};
584 cfg.reportInterval_us = interval_us;
585 cfg.batchInterval_us = batch_us;
586 cfg.sensorSpecific = 0;
587 const float clamped_sensitivity = std::clamp(sensitivity, 0.0f, 65535.0f);
588 cfg.changeSensitivity = static_cast<uint16_t>(clamped_sensitivity + 0.5f);
589 cfg.changeSensitivityEnabled = clamped_sensitivity > 0.0f;
590 int status = sh2_setSensorConfig(static_cast<sh2_SensorId_t>(sensor), &cfg);
591 if (status != SH2_OK) {
592 last_error_ = status;
593 return false;
594 }
595 return true;
596}

647template <typename CommType>
648bool BNO085<CommType>::EnterBootloader(uint32_t resetLowMs, uint32_t settleMs) noexcept {
649 if (io_.GetInterfaceType() == BNO085Interface::UARTRVC) {
650 last_error_ = SH2_ERR_BAD_PARAM;
651 return false;
652 }
653 if (state_ == BNO085DriverState::DfuInProgress) {
654 last_error_ = SH2_ERR_OP_IN_PROGRESS;
655 return false;
656 }
657 if (state_ != BNO085DriverState::Closed) {
658 Close();
659 }
660 if (state_ != BNO085DriverState::Closed) {
661 last_error_ = SH2_ERR_OP_IN_PROGRESS;
662 return false;
663 }
664
665 if (!io_.Open()) {
666 last_error_ = SH2_ERR;
667 return false;
668 }
669 // Assert BOOTN, reset, then release BOOTN (BOOTN is sampled at reset, datasheet section 1.4).
670 const bool ok = setPin(bno08x::CtrlPin::BOOTN, true) && HardwareReset(resetLowMs);
671 const bool released = setPin(bno08x::CtrlPin::BOOTN, false);
672 if (!ok || !released) {
673 io_.Close();
674 last_error_ = SH2_ERR_IO;
675 return false;
676 }
677 if (settleMs) {
678 io_.Delay(settleMs);
679 }
680 last_error_ = SH2_OK;
681 return true;
682}

832template <typename CommType>
833void BNO085<CommType>::rvcProcessByte(uint8_t c) noexcept {
834 if (rvc_frame_len_ == RVC_FRAME_LEN_) {
835 std::memmove(rvc_frame_, rvc_frame_ + 1, RVC_FRAME_LEN_ - 1);
836 rvc_frame_[RVC_FRAME_LEN_ - 1] = c;
837 } else {
838 rvc_frame_[rvc_frame_len_++] = c;
839 }
840
841 if (rvc_frame_len_ == RVC_FRAME_LEN_ && rvc_frame_[0] == 0xAA && rvc_frame_[1] == 0xAA) {
842 uint8_t check = 0;
843 for (int i = 2; i < RVC_FRAME_LEN_ - 1; ++i)
844 check += rvc_frame_[i];
845
846 if (check == rvc_frame_[RVC_FRAME_LEN_ - 1]) {
847 RvcSensorEvent event{};
848 event.timestamp_uS = io_.GetTimeUs();
849 event.index = rvc_frame_[2];
850 event.yaw = static_cast<int16_t>((rvc_frame_[4] << 8) | rvc_frame_[3]);
851 event.pitch = static_cast<int16_t>((rvc_frame_[6] << 8) | rvc_frame_[5]);
852 event.roll = static_cast<int16_t>((rvc_frame_[8] << 8) | rvc_frame_[7]);
853 event.acc_x = static_cast<int16_t>((rvc_frame_[10] << 8) | rvc_frame_[9]);
854 event.acc_y = static_cast<int16_t>((rvc_frame_[12] << 8) | rvc_frame_[11]);
855 event.acc_z = static_cast<int16_t>((rvc_frame_[14] << 8) | rvc_frame_[13]);
856 event.mi = rvc_frame_[15];
857 event.mr = rvc_frame_[16];
858
859 if (rvc_cb_) {
860 RvcSensorValue val;
861 decodeRvc(&val, &event);
862 rvc_cb_(val);
863 }
864 rvc_frame_len_ = 0;
865 }
866 }

931template <typename CommType>
932void BNO085<CommType>::dfuAppendCrc(uint8_t* packet, uint8_t len) noexcept {
933 uint16_t crc = 0xFFFF;
934 for (int n = 0; n < len; n++) {
935 uint16_t x = static_cast<uint16_t>(packet[n]) << 8;
936 for (int i = 0; i < 8; i++) {
937 if ((crc ^ x) & 0x8000)
938 crc = (crc << 1) ^ 0x1021;
939 else
940 crc = crc << 1;
941 x <<= 1;
942 }
943 }
944 packet[len] = (crc >> 8) & 0xFF;
945 packet[len + 1] = crc & 0xFF;
946}

956template <typename CommType>
957int BNO085<CommType>::dfuSend(uint8_t* p_data, uint32_t len) noexcept {
958 unsigned int retries = 0;
959 int status = SH2_OK;
960 uint8_t ack = 0;
961 bool got_ack = false;
962 uint32_t t;
963 sh2_Hal_t* hal = halWrapper_.asHal();
964
965 while (!got_ack && (retries < DFU_MAX_ATTEMPTS)) {
966 uint32_t now = hal->getTimeUs(hal);
967 uint32_t start = now;
968 status = 0;
969 while ((status == 0) && ((now - start) < DFU_SEND_TIMEOUT_US)) {
970 status = hal->write(hal, p_data, len);
971 now = hal->getTimeUs(hal);
972 }
973 if (status == 0)
974 status = SH2_ERR_TIMEOUT;
975 if (status > 0) {
976 status = 0;
977 while ((status == 0) && ((now - start) < DFU_SEND_TIMEOUT_US)) {
978 status = hal->read(hal, &ack, 1, &t);
979 now = hal->getTimeUs(hal);
980 }
981 if (status == 0)
982 status = SH2_ERR_TIMEOUT;
983 }
984 if (status > 0) {
985 if (ack == DFU_ACK) {
986 got_ack = true;
987 status = SH2_OK;
988 } else {
989 got_ack = false;
990 status = SH2_ERR_HUB;
991 }
992 }
993 if (!got_ack)
994 retries++;
995 }
996 return (status >= 0) ? SH2_OK : status;
997}

1124template <typename CommType>
1125int BNO085<CommType>::DfuWithOptions(const HcBin_t& fw, const DfuOptions& options) noexcept {
1126 if (io_.GetInterfaceType() == BNO085Interface::UARTRVC) {
1127 last_error_ = SH2_ERR_BAD_PARAM;
1128 return SH2_ERR_BAD_PARAM;
1129 }
1130 if (state_ == BNO085DriverState::RvcActive || state_ == BNO085DriverState::DfuInProgress) {
1131 last_error_ = SH2_ERR_OP_IN_PROGRESS;
1132 return SH2_ERR_OP_IN_PROGRESS;
1133 }
1134 if (state_ == BNO085DriverState::Sh2Active) {
1135 // DFU uses the same transport HAL and must not race an active SH-2 session.
1136 Close();
1137 }
1138 if (state_ != BNO085DriverState::Closed) {
1139 last_error_ = SH2_ERR_OP_IN_PROGRESS;
1140 return SH2_ERR_OP_IN_PROGRESS;
1141 }
1142 state_ = BNO085DriverState::DfuInProgress;
1143 prepareHalWrapper();
1144
1145 int rc, status = SH2_OK;
1146 uint32_t app_len = 0;
1147 uint8_t packet_len = 0;
1148 uint32_t offset = 0;
1149 const char* meta = nullptr;
1150 bool fw_opened = false;
1151 bool hal_opened = false;
1152 uint8_t app_address = 0;
1153 bool address_switched = false;
1154 uint8_t dfu_buff[DFU_MAX_PACKET_LEN + 2];
1155 sh2_Hal_t* hal = halWrapper_.asHal();
1156
1157 rc = fw.open();
1158 if (rc != 0) {
1159 status = SH2_ERR;
1160 goto dfu_end;
1161 }
1162 fw_opened = true;
1163
1164 if (options.requireFormatMatch) {
1165 const char* expected_format = options.requiredFormat ? options.requiredFormat : "BNO_V1";
1166 meta = fw.getMeta("FW-Format");
1167 if (!meta || std::strcmp(meta, expected_format) != 0) {
1168 status = SH2_ERR_BAD_PARAM;
1169 goto dfu_close;
1170 }
1171 }
1172
1173 if (options.requirePartNumber) {
1174 meta = fw.getMeta("SW-Part-Number");
1175 if (!meta) {
1176 status = SH2_ERR_BAD_PARAM;
1177 goto dfu_close;
1178 }
1179 if (options.requiredPartNumber) {
1180 if (std::strcmp(meta, options.requiredPartNumber) != 0) {
1181 status = SH2_ERR_BAD_PARAM;
1182 goto dfu_close;
1183 }
1184 } else if (!dfuIsKnownPartNumber(meta)) {
1185 status = SH2_ERR_BAD_PARAM;
1186 goto dfu_close;
1187 }
1188 }
1189
1190 app_len = fw.getAppLen();
1191 if (app_len < 1024) {
1192 status = SH2_ERR_BAD_PARAM;
1193 goto dfu_close;
1194 }
1195
1196 packet_len = fw.getPacketLen();
1197 if (options.packetLenOverride != 0) {
1198 packet_len = options.packetLenOverride;
1199 }
1200 if (packet_len == 0 || packet_len > DFU_MAX_PACKET_LEN)
1201 packet_len = DFU_MAX_PACKET_LEN;
1202
1203 if (io_.GetInterfaceType() == BNO085Interface::I2C) {
1204 // Datasheet section 1.4: over I2C the bootloader answers at 0x28 / 0x29, not 0x4A / 0x4B.
1205 status = dfuSelectBootloaderAddress(app_address, address_switched);
1206 if (status != SH2_OK)
1207 goto dfu_close;
1208 }
1209
1210 status = hal->open(hal);
1211 if (status != SH2_OK)
1212 goto dfu_close;
1213 hal_opened = true;
1214
1215 status = dfuSendAppSize(dfu_buff, app_len);
1216 if (status != SH2_OK)
1217 goto dfu_close;
1218 status = dfuSendPktSize(dfu_buff, packet_len);
1219 if (status != SH2_OK)
1220 goto dfu_close;
1221
1222 if (options.progress) {
1223 options.progress(DfuProgress{0, app_len});
1224 }
1225
1226 offset = 0;
1227 while (offset < app_len) {
1228 uint32_t to_send = app_len - offset;
1229 if (to_send > packet_len)
1230 to_send = packet_len;
1231 status = fw.getAppData(dfu_buff, offset, to_send);
1232 if (status != SH2_OK) {
1233 status = SH2_ERR;
1234 goto dfu_close;
1235 }
1236 status = dfuSendPkt(dfu_buff, dfu_buff, to_send);
1237 if (status != SH2_OK)
1238 goto dfu_close;
1239 offset += to_send;
1240 if (options.progress) {
1241 options.progress(DfuProgress{offset, app_len});
1242 }
1243 }
1244
1245dfu_close:
1246 if (fw_opened)
1247 fw.close();
1248 if (status == SH2_OK && hal_opened) {
1249 uint32_t now = hal->getTimeUs(hal), start = now;
1250 while ((now - start) < DFU_DELAY_POST_US)
1251 now = hal->getTimeUs(hal);
1252 }
1253 if (hal_opened)
1254 hal->close(hal);
1255 if (address_switched && !dfuRestoreAppAddress(app_address) && status == SH2_OK)
1256 status = SH2_ERR_IO;
1257
1258dfu_end:
1259 state_ = BNO085DriverState::Closed;
1260 last_error_ = status;
1261 return status;
1262}
