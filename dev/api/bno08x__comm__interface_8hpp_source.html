103 using Result = GpioSetResult<Port>;
104 static_assert(std::is_void_v<Result> || kGpioSetReportsStatus<Port>,
105 "bno08x: the port's GpioSet(pin, signal) must return bool (true = pin driven, "
106 "false = GPIO error), or void for a port that never reports errors");
107 if constexpr (std::is_void_v<Result>) {
